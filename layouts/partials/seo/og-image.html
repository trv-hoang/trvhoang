{{- /* Share image for a page, as dict url/width/height/alt (always 1200×630 raster). First match wins: 1. blog post with a raster coverImage in its bundle → that photo, cropped; 2. any other blog post → base card with the post title drawn on it; 3. every other page → site card with avatar, name and role. Shared /vi/ copies use the English page, so both copies share one image URL. */ -}} {{- $page := partial "seo/canonical.html" . -}} {{- $name := site.Params.author.name -}} {{- $font := resources.Get "fonts/BeVietnamPro-SemiBold.ttf" -}} {{- $base := resources.Get "images/og/base.png" -}} {{- $white := "#fafaf9" -}} {{- $amber := "#f59e0b" -}} {{- $img := "" -}} {{- $alt := $name -}} {{- if partial "seo/is-shared.html" $page -}} {{- $alt = $page.Title -}} {{- with $page.Params.coverImage -}} {{- with $page.Resources.Get (path.Base .) -}} {{- if in (slice "image/jpeg" "image/png" "image/webp") .MediaType.Type -}} {{- /* Always JPEG: not every link-preview service shows WebP. */ -}} {{- $img = .Fill "1200x630 Smart q82 jpg" -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $img -}} {{- $lines := partial "seo/wrap-title.html" $page.Title -}} {{- $img = $base | images.Filter (images.Text (delimit $lines "\n") (dict "font" $font "color" $white "size" 64 "linespacing" 14 "x" 96 "y" 120)) (images.Text $name (dict "font" $font "color" $amber "size" 30 "x" 96 "y" 520)) -}} {{- end -}} {{- else -}} {{- /* ogImage may also be a full URL (see hugo.toml): then there is no local avatar to draw, and the text moves to the left edge. */ -}} {{- $filters := slice -}} {{- $x := 96 -}} {{- with resources.Get site.Params.ogImage -}} {{- $filters = $filters | append (images.Overlay (.Fill "360x360 Smart") 96 135) -}} {{- $x = 512 -}} {{- end -}} {{- $filters = $filters | append (images.Text $name (dict "font" $font "color" $white "size" 64 "x" $x "y" 220)) (images.Text (site.Home.Params.role | default "") (dict "font" $font "color" $amber "size" 34 "x" $x "y" 320)) -}} {{- $img = $base | images.Filter $filters -}} {{- end -}} {{- return dict "url" $img.Permalink "width" $img.Width "height" $img.Height "alt" $alt -}}
