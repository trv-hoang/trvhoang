{{/* Floating navigation. Items come from [menus] in hugo.toml; an item's params.match lists extra path prefixes that count as "inside" it (project pages live under /project/, work pages under /about/work/). */}} {{- $here := .RelPermalink -}} {{- $current := "" -}} {{- range site.Menus.main -}} {{- $item := . -}} {{- if or (and (eq .URL "/") (eq $here "/")) (and (ne .URL "/") (strings.HasPrefix $here .URL)) }}{{ $current = $item.Identifier }}{{ end -}} {{- range .Params.match }}{{ if strings.HasPrefix $here . }}{{ $current = $item.Identifier }}{{ end }}{{ end -}} {{- end -}}
{{- range site.Menus.main }} {{- $active := eq .Identifier $current }} {{ if $active }}{{ end }} {{ .Name }} {{- end }}
{{- $label := i18n "menu" -}} {{- range site.Menus.main }}{{ if eq .Identifier $current }}{{ $label = .Name }}{{ end }}{{ end }} {{ $label }}
{{/* the same page in the other language, or that language's home page */}} {{- $switch := "" -}} {{- range .Translations }}{{ $switch = .RelPermalink }}{{ end -}} {{- if not $switch }}{{ range site.Home.Translations }}{{ $switch = .RelPermalink }}{{ end }}{{ end -}} {{ with $switch }} {{ i18n "language_short" }} {{ end }}
{{- range $i, $item := site.Menus.main }} {{ .Name }} {{- end }}
